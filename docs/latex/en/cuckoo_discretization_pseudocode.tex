\subsection{Discretization and Pseudocode}
\label{subsec:cs-discretization}

In the continuous formulation of Yang and Deb, a new solution
$\mathbf{x}_i^{t+1}$ for cuckoo $i$ is generated by a L\'evy flight,
\begin{equation}
  \mathbf{x}_i^{t+1} \;=\; \mathbf{x}_i^{t} \;+\; \alpha \oplus
  \mathrm{L\acute{e}vy}(\lambda),
  \label{eq:cs-levy-update}
\end{equation}
where $\alpha > 0$ is a step-size scale and the random step length is drawn
from the heavy-tailed L\'evy distribution
$\mathrm{L\acute{e}vy} \sim u = t^{-\lambda}$ with $1 < \lambda \leq 3$.
Neither the vector addition nor the entrywise product $\oplus$ is defined on
the discrete space of CARP solutions (sets of routes over required edges), so
Eq.~\eqref{eq:cs-levy-update} is discretized as follows:

\begin{itemize}
  \item \textbf{Step length $\to$ number of local moves.} A L\'evy step of
    length $u$ is mapped to a burst of $L$ consecutive local moves applied to
    the current solution, where $L$ inherits the heavy tail of the
    distribution:
    \begin{equation}
      x \sim |\mathcal{N}(0,1)|, \qquad
      L \;=\; \min\!\bigl(12,\; 1 + \lfloor x^{1/\lambda} \cdot
      P_{\mathrm{base}} \rfloor\bigr),
      \label{eq:cs-discrete-levy}
    \end{equation}
    so most flights perform one or two moves while occasional flights chain
    many, reproducing the short-steps/rare-long-jumps profile of the L\'evy
    walk. The cap $L \leq 12$ bounds the disruption of a single flight.
  \item \textbf{Scale $\alpha$ $\to$ $P_{\mathrm{base}}$.} The role of the
    continuous step size $\alpha$ is played by the discrete scale
    $P_{\mathrm{base}}$, set instance-aware to
    $\max(3, \lfloor \sqrt{n}/2 \rfloor)$ for $n$ required tasks and
    multiplied by the tunable factor $f_{\mathrm{steps}}$.
  \item \textbf{Move vocabulary.} Each of the $L$ moves applies one of the
    nine shared neighborhood operators
    (Table~\ref{tab:neighborhood-operators}), drawn uniformly from the
    intra-route or inter-route group; the group is selected with probability
    $P(\text{inter}) = p_{inter}$, raised to $\max(p_{inter}, 0.8)$ whenever
    the solution violates capacity.
  \item \textbf{Objective.} Solutions are compared with the penalized cost
    $f(s) = c(s) + \omega \cdot \mathrm{viol}(s)$, where $c(s)$ is the
    routing cost, $\mathrm{viol}(s)$ the total capacity excess, and
    $\omega = \max(10 \cdot \mathrm{median}(d),\, 10)$ a capacity-penalty
    weight derived from the distance matrix $d$.
\end{itemize}

The remaining components of the canonical algorithm carry over directly: each
cuckoo competes against a randomly chosen nest (greedy replacement), and a
fraction $p_a$ of the worst nests is abandoned and rebuilt by L\'evy flights
from the current best nest. Nests are initialized from a Path-Scanning warm
start (best of five constructions). Algorithm~\ref{alg:cs-carp} summarizes
the procedure; the calibrated setting on the benchmark corpus is
$p_a = 0.15$, $\lambda = 1.3$, $K = 25$ nests, and
$f_{\mathrm{steps}} = 0.25$.

\begin{algorithm}[htbp]
\DontPrintSemicolon
\SetKwProg{Fn}{Function}{:}{}
\SetKwFunction{LevyFlight}{L\'evyFlight}
\caption{Discrete Cuckoo Search for the CARP}
\label{alg:cs-carp}

\KwIn{CARP instance; nests $K$; abandonment fraction $p_a$; L\'evy exponent
      $\lambda$; scale $P_{\mathrm{base}}$; inter-route bias $p_{inter}$;
      penalty weight $\omega$; max evaluations $10^6$ or max time 300\,s}
\KwOut{Best solution found $s^{*}$}

\BlankLine
\Fn{\LevyFlight{$s$, $\mathcal{O}$}}{
  draw $x \sim |\mathcal{N}(0,1)|$;\quad
  $L \leftarrow \min\!\bigl(12,\, 1 + \lfloor x^{1/\lambda}
      P_{\mathrm{base}} \rfloor\bigr)$\;
  \lFor{$\ell \leftarrow 1$ \KwTo $L$}{
      $s \leftarrow$ random move on $s$ with an operator of $\mathcal{O}$}
  \Return $s$\;
}

\BlankLine
initialize nests $N[0..K{-}1]$ from a Path-Scanning warm start;\quad
$s^{*} \leftarrow \arg\min_i f(N[i])$\;
\While{evals $< 10^6$ \textbf{and} time $< 300$\,s \textbf{and} not stagnated}{
  $\mathcal{O} \leftarrow$ inter-route group with prob.\
  $\max(p_{inter}, 0.8 \cdot \mathbb{1}[\text{violation}])$, intra-route
  otherwise\;
  \tcp{L\'evy generation and greedy competition}
  \For{$i \leftarrow 0$ \KwTo $K-1$}{
    $s' \leftarrow$ \LevyFlight{$N[i]$, $\mathcal{O}$};\quad
    $j \leftarrow$ random nest index\;
    \lIf{$f(s') < f(N[j])$}{$N[j] \leftarrow s'$}
    \lIf{$f(s') < f(s^{*})$}{$s^{*} \leftarrow s'$}
  }
  \tcp{abandonment of the worst nests}
  \ForEach{$k$ in the $\lfloor p_a K \rfloor$ worst nests}{
    $N[k] \leftarrow$ \LevyFlight{$\arg\min_i f(N[i])$, $\mathcal{O}$};\quad
    \lIf{$f(N[k]) < f(s^{*})$}{$s^{*} \leftarrow N[k]$}
  }
}
\Return $s^{*}$\;
\end{algorithm}
